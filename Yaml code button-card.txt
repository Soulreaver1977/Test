  status: |
    [[[
      const e = states['person.jarvis'];

      if (e?.state === 'home') {

          ">●</span>

      `;

    [[[
      const geo = states['sensor.iphone_van_edwin_geocoded_location'];

      const locality =
        geo?.attributes?.Locality ||
        geo?.attributes?.locality;

      return `
        <span style="
          display:inline-block;
          width:24px;
          text-align:center;
        ">📍</span>
        <span>${locality || 'Locatie onbekend'}</span>
      `;
    ]]]
  battery: |
    [[[
      const e = states['sensor.iphone_van_edwin_battery_level'];
      const battery = Number(e?.state);

      if (isNaN(battery)) {
        return '';
      }

      let color = '#35e88a';

      if (battery < 20) {
        color = '#ff5252';
      } else if (battery < 50) {

            text-align:center;

          <span style="color:#35e88a">Aan het laden</span>

        <span style="color:rgba(255,255,255,0.55)">Niet laden</span>
      `;
    ]]]

            style="

              height:16px;
